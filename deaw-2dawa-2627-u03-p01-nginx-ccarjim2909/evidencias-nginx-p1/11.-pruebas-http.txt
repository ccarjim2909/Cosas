  <h2>Repetidor de Hola Mundo</h2>
  <input type="number" id="cantidadInput" min="1" max="100" placeholder="Ej. 5">
  <button onclick="repetir()">Generar</button>
  <div id="salida"></div>

  <script>
    function repetir() {
      const cantidad = parseInt(document.getElementById('cantidadInput').value);
      const contenedor = document.getElementById('salida');
      contenedor.innerHTML = "";
      if (isNaN(cantidad) || cantidad <= 0) {
        contenedor.textContent = "Por favor, introduce un número válido mayor que 0.";
        return;
      }
      for (let i = 0; i < cantidad; i++) {
        const linea = document.createElement('div');
        linea.textContent = (i + 1) + ". ¡Hola mundo!";
        contenedor.appendChild(linea);
      }
    }
  </script>
</body>
</html>
